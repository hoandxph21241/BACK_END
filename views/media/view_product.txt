      <div id="uploadProgressBar" class="progress-bar progress-bar-striped bg-info progress-bar-animated"
        role="progressbar" style="width: 0%;" aria-valuemin="0" aria-valuemax="100">0%</div>
    </div>

    <!-- Alert Messages -->
    <div id="alertContainer"></div>

    <!-- Upload Form -->
    <form id="uploadForm">
      <div class="mb-3">
        <label for="mp3Name" class="form-label">Name:</label>
        <input type="text" class="form-control" id="mp3Name" name="name" placeholder="Nhập tên bài hát (tùy chọn)">
      </div>

      <div class="mb-3">
        <label for="categoryId" class="form-label">Category:</label>
        <select class="form-select" id="categoryId" name="categoryId" required>
          <option value="">-- Chọn danh mục --</option>
          <% if (listidCategory && listidCategory.length > 0) { %>

      <button type="submit" class="btn btn-primary" id="uploadBtn">
        <span class="spinner-border spinner-border-sm d-none" role="status" aria-hidden="true"></span>

  <!-- AJAX Upload Script -->
  <script>
// Sử dụng function để có thể gọi lại khi cần
function initUploadForm() {
  const uploadForm = document.getElementById('uploadForm');
  const uploadBtn = document.getElementById('uploadBtn');
  const backBtn = document.getElementById('backBtn');
  const progressContainer = document.getElementById('uploadProgressContainer');
  const progressBar = document.getElementById('uploadProgressBar');
  const alertContainer = document.getElementById('alertContainer');
  const spinner = uploadBtn.querySelector('.spinner-border');

  // Ngăn chặn form submit mặc định và xử lý bằng AJAX
  uploadForm.addEventListener('submit', function(event) {
    event.preventDefault();
    event.stopPropagation();
    console.log('Form submitted - handling with AJAX');
    handleFormSubmit();
  });

  // Ngăn chặn click button submit mặc định
  uploadBtn.addEventListener('click', function(event) {
    event.preventDefault();
    console.log('Upload button clicked');
    handleFormSubmit();
  });

  // Handle back button
  backBtn.addEventListener('click', function(event) {
    event.preventDefault();
    if (window.parent !== window && window.parent.document.getElementById('main')) {
      loadHomePage();
    } else {
      window.location.href = '/home';
    }
  });

  // Function to load home page in SPA mode
  function loadHomePage() {
    fetch('/home')
      .then(res => res.text())
      .then(html => {
        if (window.parent && window.parent.document) {
          const mainElement = window.parent.document.getElementById('main');
          if (mainElement) {
            mainElement.innerHTML = html;
            window.parent.history.pushState(null, '', '/home');
          }
        }
      })
      .catch(() => {
        window.location.href = '/home';
      });
  }

  // Show alert function
  function showAlert(message, type = 'danger') {
    alertContainer.innerHTML = `
      <div class="alert alert-${type} alert-dismissible fade show" role="alert">
        ${message}
        <button type="button" class="btn-close" data-bs-dismiss="alert" aria-label="Close"></button>
      </div>
    `;
  }

  // Validate file size (50MB max)
  document.getElementById('mp3File').addEventListener('change', function(e) {
    const file = e.target.files[0];
    if (file) {
      const maxSize = 50 * 1024 * 1024; // 50MB
      if (file.size > maxSize) {
        showAlert('File quá lớn! Vui lòng chọn file nhỏ hơn 50MB.');
        e.target.value = '';
        return;
      }
      
      // Clear any previous alerts
      alertContainer.innerHTML = '';
    }
  });

  // Main form submission handler
  function handleFormSubmit() {
    console.log('handleFormSubmit called');
    
    // Validate form
    const categoryId = document.getElementById('categoryId').value;
    const mp3File = document.getElementById('mp3File').files[0];
    
    if (!categoryId) {
      showAlert('Vui lòng chọn danh mục!');
      return false;
    }
    
    if (!mp3File) {
      showAlert('Vui lòng chọn file MP3!');
      return false;
    }

    // Prepare form data
    const formData = new FormData();
    formData.append('name', document.getElementById('mp3Name').value);
    formData.append('categoryId', categoryId);
    formData.append('mp3File', mp3File);

    console.log('Starting upload...');

    // Show progress bar và disable buttons
    alertContainer.innerHTML = '';
    progressContainer.style.display = 'block';
    progressBar.style.width = '0%';
    progressBar.innerText = '0%';
    progressBar.className = 'progress-bar progress-bar-striped bg-info progress-bar-animated';
    
    uploadBtn.disabled = true;
    backBtn.disabled = true;
    spinner.classList.remove('d-none');

    // Create XMLHttpRequest
    const xhr = new XMLHttpRequest();
    
    // Upload progress tracking
    xhr.upload.addEventListener('progress', function (e) {
      if (e.lengthComputable) {
        const percent = Math.round((e.loaded / e.total) * 100);
        progressBar.style.width = percent + '%';
        progressBar.innerText = percent + '%';
        progressBar.setAttribute('aria-valuenow', percent);
        console.log(`Upload progress: ${percent}%`);
      }
    });

    // Handle upload start
    xhr.upload.addEventListener('loadstart', function(e) {
      console.log('Upload started');
      progressBar.style.width = '1%';
      progressBar.innerText = '1%';
      progressBar.setAttribute('aria-valuenow', 1);
    });

    // Handle upload completion
    xhr.upload.addEventListener('load', function(e) {
      console.log('Upload completed');
      progressBar.style.width = '100%';
      progressBar.innerText = '100%';
      progressBar.setAttribute('aria-valuenow', 100);
    });

    // Handle response
    xhr.onreadystatechange = function () {
      if (xhr.readyState === XMLHttpRequest.DONE) {
        console.log('Request completed with status:', xhr.status);
        
        // Re-enable buttons
        uploadBtn.disabled = false;
        backBtn.disabled = false;
        spinner.classList.add('d-none');
        
        if (xhr.status >= 200 && xhr.status < 300) {
          try {
            const response = JSON.parse(xhr.responseText);
            console.log('Server response:', response);
            
            if (response.success) {
              // Success - update progress bar
              progressBar.classList.remove('bg-info', 'progress-bar-animated');
              progressBar.classList.add('bg-success');
              progressBar.style.width = '100%';
              progressBar.innerText = '100% - Hoàn thành!';
              progressBar.setAttribute('aria-valuenow', 100);
              
              showAlert('Upload thành công! Đang chuyển hướng...', 'success');
              
              // Redirect after 2 seconds
              setTimeout(() => {
                if (window.parent !== window && window.parent.document.getElementById('main')) {
                  loadHomePage();
                } else {
                  window.location.href = '/home';
                }
              }, 2000);
            } else {
              // Server returned error
              progressBar.classList.remove('bg-info', 'progress-bar-animated');
              progressBar.classList.add('bg-danger');
              progressBar.style.width = '100%';
              progressBar.innerText = 'Lỗi!';
              progressBar.setAttribute('aria-valuenow', 100);
              
              showAlert(response.error || response.message || 'Upload thất bại!');
            }
          } catch (e) {
            // JSON parsing error
            console.error('JSON parsing error:', e);
            console.error('Response text:', xhr.responseText);
            
            progressBar.classList.remove('bg-info', 'progress-bar-animated');
            progressBar.classList.add('bg-danger');
            progressBar.style.width = '100%';
            progressBar.innerText = 'Lỗi!';
            progressBar.setAttribute('aria-valuenow', 100);
            
            showAlert('Có lỗi xảy ra trong quá trình xử lý phản hồi!');
          }
        } else {
          // HTTP error
          console.error('HTTP error:', xhr.status, xhr.responseText);
          
          progressBar.classList.remove('bg-info', 'progress-bar-animated');
          progressBar.classList.add('bg-danger');
          progressBar.style.width = '100%';
          progressBar.innerText = 'Lỗi!';
          progressBar.setAttribute('aria-valuenow', 100);
          
          try {
            const response = JSON.parse(xhr.responseText);
            showAlert(response.error || response.message || `Upload thất bại! (Mã lỗi: ${xhr.status})`);
          } catch (e) {
            showAlert(`Upload thất bại! (Mã lỗi: ${xhr.status})`);
          }
        }
      }
    };

    // Handle network errors
    xhr.onerror = function() {
      console.error('Network error occurred');
      
      uploadBtn.disabled = false;
      backBtn.disabled = false;
      spinner.classList.add('d-none');
      
      progressBar.classList.remove('bg-info', 'progress-bar-animated');
      progressBar.classList.add('bg-danger');
      progressBar.style.width = '100%';
      progressBar.innerText = 'Lỗi kết nối!';
      progressBar.setAttribute('aria-valuenow', 100);
      
      showAlert('Lỗi kết nối! Vui lòng kiểm tra mạng và thử lại.');
    };

    // Handle timeout
    xhr.ontimeout = function() {
      console.error('Request timeout');
      
      uploadBtn.disabled = false;
      backBtn.disabled = false;
      spinner.classList.add('d-none');
      
      progressBar.classList.remove('bg-info', 'progress-bar-animated');
      progressBar.classList.add('bg-danger');
      progressBar.style.width = '100%';
      progressBar.innerText = 'Timeout!';
      progressBar.setAttribute('aria-valuenow', 100);
      
      showAlert('Upload timeout! Vui lòng thử lại.');
    };

    // Configure and start upload
    xhr.open('POST', '/product/upload');
    xhr.timeout = 300000; // 5 minutes timeout
    xhr.send(formData);
    
    return false;
  }
}

// Khởi tạo khi DOM ready
if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', initUploadForm);
} else {
  // Nếu DOM đã sẵn sàng (trường hợp load động)
  initUploadForm();
}

// Export function để có thể gọi từ parent window
if (typeof window !== 'undefined') {
  window.initUploadForm = initUploadForm;
}
</script>
