const Model = require("../../model/db_song");
const bcrypt = require("bcrypt");
const jwt = require("jsonwebtoken");
const crypto = require("crypto");
const { token } = require("morgan");

class AuthService {
  async register(req) {
    try {
      const { nameAccount, namePassword, confirmPassword, fullName, gmail } =
        req.body;
      if (!nameAccount || !namePassword) {
        return {
          success: false,
          message: "Please enter account and password.",
        };
      }
      if (namePassword !== confirmPassword) {
        return {
          success: false,
          message: "Password confirmation does not match.",
        };
      }
      const exists = await Model.UserModel.findOne({
        nameAccount,
      });

      if (exists) {
        return {
          success: false,
          message: "Account already exists.",
        };
      }
      const hashPassword = await bcrypt.hash(namePassword, 10);
      const user = await Model.UserModel.create({
        userID: crypto.randomUUID(),
        nameAccount,
        namePassword: hashPassword,
        fullName: fullName ?? "",
        gmail: gmail ?? "",
        refreshTokens: [],
        role: 1,
      });
      return {
        success: true,
        message: "Register successfully.",
        data: {
          id: user._id,
          userID: user.userID,
          account: user.nameAccount,
        },
      };
    } catch (err) {
      return {
        success: false,
        message: err.message,
      };
    }
  }

  async login(req) {
    try {
      console.log(process.env.JWT_ACCESS_EXPIRES);
      console.log(process.env.JWT_REFRESH_EXPIRES);
      const { nameAccount, namePassword } = req.body;
      if (!nameAccount || !namePassword) {
        return {
          success: false,
          message: "Please enter account and password.",
        };
      }
      const user = await Model.UserModel.findOne({
        nameAccount,
      });
      if (!user) {
        return {
          success: false,
          message: "Account does not exist.",
        };
      }
      const isMatch = await bcrypt.compare(namePassword, user.namePassword);
      if (!isMatch) {
        return {
          success: false,
          message: "Incorrect password.",
        };
      }
      const accessToken = jwt.sign(
        {
          id: user._id,
          userID: user.userID,
          role: user.role,
        },
        process.env.JWT_SECRET,
        {
          expiresIn: process.env.JWT_ACCESS_EXPIRES,
        },
      );
      const refreshToken = jwt.sign(
        {
          id: user._id,
          userID: user.userID,
        },
        process.env.JWT_SECRET,
        {
          expiresIn: process.env.JWT_REFRESH_EXPIRES,
        },
      );
      user.refreshTokens = user.refreshTokens.filter(
        (item) => item.expiresAt > new Date(),
      );
      user.refreshTokens.push({
        token: refreshToken,
        device: req.headers["user-agent"] || "Unknown Device",
        createdAt: new Date(),
        expiresAt: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000),
      });
      await user.save();
      return {
        success: true,
        message: "Login successfully.",
        accessToken,
        refreshToken,
        data: {
          id: user._id,
          userID: user.userID,
          account: user.nameAccount,
          fullName: user.fullName,
          avatar: user.imageAccount,
          gmail: user.gmail,
          role: user.role,
        },
      };
    } catch (err) {
      return {
        success: false,
        message: err.message,
      };
    }
  }

  async refreshToken(req) {
    try {
      const { refreshToken } = req.body;
      if (!refreshToken) {
        return {
          success: false,
          message: "Refresh token is required.",
        };
      }
      let payload;
      try {
        payload = jwt.verify(refreshToken, process.env.JWT_SECRET);
      } catch (err) {
        return {
          success: false,
          message: "Refresh token is invalid.",
        };
      }
      const user = await Model.UserModel.findById(payload.id);
      if (!user) {
        return {
          success: false,
          message: "User not found.",
        };
      }
      const exists = user.refreshTokens.find(
        (item) => item.token === refreshToken,
      );
      if (!exists) {
        return { success: false, message: "Refresh token is invalid." };
      }
      const accessToken = jwt.sign(
        {
          id: user._id,
          userID: user.userID,
          role: user.role,
        },
        process.env.JWT_SECRET,
        {
          expiresIn: process.env.JWT_ACCESS_EXPIRES,
        },
      );
      const newRefreshToken = jwt.sign(
        {
          id: user._id,
        },
        process.env.JWT_SECRET,
        {
          expiresIn: process.env.JWT_EXPIRES,
        },
      );
      return {
        success: true,
        message: "Refresh token successfully.",
        accessToken,
        refreshToken,
      };
    } catch (err) {
      return {
        success: false,
        message: err.message,
      };
    }
  }

  async logout(req) {
    try {
      const { refreshToken } = req.body;
      if (!refreshToken) {
        return {
          success: false,
          message: "Refresh token is required.",
        };
      }
      const user = await Model.UserModel.findOne({
        "refreshTokens.token": refreshToken,
      });
      if (!user) {
        return {
          success: false,
          message: "User not found.",
        };
      }
      user.refreshTokens = user.refreshTokens.filter(
        (item) => item.token !== refreshToken,
      );
      await user.save();
      return {
        success: true,
        message: "Logout successfully.",
      };
    } catch (err) {
      return {
        success: false,
        message: err.message,
      };
    }
  }

  async googleLogin(req, res, next) {
    return passport.authenticate("google", {
      scope: ["profile", "email"],
      session: false,
    })(req, res, next);
  }

  async googleCallback(req) {
    try {
      const googleUser = req.user;
      if (!googleUser) {
        return {
          success: false,
          message: "Google authentication failed.",
        };
      }
      let user = await Model.UserModel.findOne({
        gmail: googleUser.gmail,
      });
      if (!user) {
        user = await Model.UserModel.create({
          userID: crypto.randomUUID(),
          nameAccount: googleUser.gmail,
          namePassword: "",
          gmail: googleUser.gmail,
          fullName: googleUser.fullName,
          imageAccount: googleUser.imageAccount,
          Authorization: "",
          refreshTokens: [],
          role: 1,
        });
      }
      const accessToken = jwt.sign(
        {
          id: user._id,
          userID: user.userID,
          role: user.role,
        },
        process.env.JWT_SECRET,
        {
          expiresIn: process.env.JWT_ACCESS_EXPIRES,
        },
      );
      const refreshToken = jwt.sign(
        {
          id: user._id,
        },
        process.env.JWT_SECRET,
        {
          expiresIn: process.env.JWT_EXPIRES,
        },
      );
      user.refreshTokens.push({
        token: refreshToken,
        device: req.headers["user-agent"] || "Google Login",
        createdAt: new Date(),
        expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000), // 7 days
      });
      await user.save();
      return {
        success: true,
        message: "Google login successfully.",
        accessToken,
        refreshToken,
        data: {
          id: user._id,
          userID: user.userID,
          account: user.nameAccount,
          fullName: user.fullName,
          gmail: user.gmail,
          avatar: user.imageAccount,
          role: user.role,
        },
      };
    } catch (err) {
      return {
        success: false,
        message: err.message,
      };
    }
  }

  async profile(req) {
    try {
      const payload = req.user;
      const user = await Model.UserModel.findById(payload.id);
      if (!user) {
        return {
          success: false,
          message: "User not found.",
        };
      }
      return {
        success: true,
        message: "Profile loaded successfully.",
        data: {
          id: user._id,
          userID: user.userID,
          account: user.nameAccount,
          fullName: user.fullName,
          gmail: user.gmail,
          avatar: user.imageAccount,
          role: user.role,
        },
      };
    } catch (err) {
      return {
        success: false,
        message: err.message,
      };
    }
  }
}

module.exports = new AuthService();
